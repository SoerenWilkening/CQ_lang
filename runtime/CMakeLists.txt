# runtime/CMakeLists.txt — trace-only runtime.
#
# Builds libcq_runtime as a static library (PRD §6: trace-only stubs,
# e2e tests link the host binary against this). C11, propagates -Wall
# -Wextra from the top-level CMakeLists.txt.

add_library(cq_runtime STATIC
    cq_runtime.c
)

target_include_directories(cq_runtime PUBLIC
    ${CMAKE_CURRENT_SOURCE_DIR}
)

add_library(cq_templates STATIC
    cq_templates.c
    cq_intrinsic_templates.c   # Phase 7.1 Increment E — curated reversible llvm.* idioms
    cq_libm_templates.c        # bd 3c6m.2 — curated direct-libm math.h functions
)

target_include_directories(cq_templates PUBLIC
    ${CMAKE_CURRENT_SOURCE_DIR}
)

target_link_libraries(cq_templates PUBLIC cq_runtime)

# test_trace drives the API and the ctest target diffs its stdout
# against the golden trace, pinning the format before the IR pass or
# e2e fixtures begin to consume it.
add_executable(runtime_trace_test test_trace.c)
target_link_libraries(runtime_trace_test PRIVATE cq_templates cq_runtime)

add_test(
    NAME runtime_trace_test
    COMMAND bash -c "$<TARGET_FILE:runtime_trace_test> | diff -u - ${CMAKE_CURRENT_SOURCE_DIR}/test_trace.expected.log"
)

# The generated cq_templates.h carries 680 extern decls but only 14
# have bodies today (the hand-written stubs); the rest land later. To
# verify the header itself parses clean under the strict warning set
# without forcing a link step, we build test_header_compile.c into an
# OBJECT library — compilation is exercised, linking is not. The ctest
# entry drives `cmake --build` against that object library, so the test
# passes iff the header compiles warning-free under -Wall -Wextra
# -Wpedantic -Werror. The strict flags live only on this target (root
# CMakeLists ships -Wall -Wextra without -Werror, per its comment on
# line 9-11).
add_library(runtime_header_compile_obj OBJECT test_header_compile.c)
target_include_directories(runtime_header_compile_obj PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}
)
target_compile_options(runtime_header_compile_obj PRIVATE
    -Wall -Wextra -Wpedantic -Werror
)

# Aggregate convenience target so callers can `cmake --build . --target
# runtime_header_compile` to force a rebuild before running ctest.
add_custom_target(runtime_header_compile
    DEPENDS runtime_header_compile_obj
)

add_test(
    NAME runtime_header_compile
    COMMAND ${CMAKE_COMMAND} --build ${CMAKE_BINARY_DIR}
            --target runtime_header_compile_obj
)

# bd 0ktq — the LINK-layer witness, the sibling half of runtime_header_compile
# above. That gate deliberately builds into an OBJECT library so the linker never
# resolves anything: it proves the headers PARSE. Nothing proved that the archive
# actually DEFINES what those headers declare.
#
# MEASURED GAP (2026-07-27, the bd gowy residual): of the generated surface, only
# 376 symbols were witnessed by any layer (e2e goldens + ir-pass lit) and 2103
# were witnessed nowhere. Per Rule 2 the e2e golden diff is "the only layer that
# LINKS and proves every emitted symbol resolves and signature-matches" — so most
# of the surface about to be FROZEN had no such proof. Chasing it with authored
# per-symbol e2e slices is the wrong fix (a fictional program, the exact thing bd
# tkju/dbab515 removed, in a disjoint scoreboard category).
#
# cq_link_smoke.c is GENERATED by tools/gen_templates.py from the same row
# enumeration that emits the bodies, so it cannot enumerate a different set than
# it defines, and it rides `--check`: adding a YAML row drifts the committed file
# and forces a regeneration that necessarily covers the new symbol.
#
# WHAT PASSING MEANS: all 2880 generated cq_template_* symbols are defined in
# libcq_templates.a and resolve at link. It is NOT a coverage number and NOT a
# correctness claim — the binary calls nothing. It is a REGRESSION gate: measured
# at introduction, 2880 declared / 2880 defined / 0 missing, so it passes on day
# one by construction and exists to catch the day it stops.
#
# NOT covered by this gate, deliberately: the hand-written cqrt_* core ABI (not
# YAML-generated, so enumerating it here would mean parsing a hand-written header
# from a YAML-driven emitter).
add_executable(cq_link_smoke cq_link_smoke.c)
target_link_libraries(cq_link_smoke PRIVATE cq_templates cq_runtime)

add_test(NAME runtime_link_smoke COMMAND cq_link_smoke)

# Mutant validation for the gate above (the tools/free_pairing_check_test.py
# discipline). runtime_link_smoke passes today BY CONSTRUCTION, so green proves
# nothing about whether it can still fail; this asserts it does fail when it
# should. M2 is the load-bearing mutant: MEASURED, a `static` unreferenced table
# at -O2 is discarded before the linker resolves anything, so the gate would go
# silently vacuous — which is why emit_smoke gives the table EXTERNAL linkage.
find_program(_RT_PYTHON NAMES python3 python REQUIRED)

add_test(
    NAME runtime_link_smoke_selftest
    COMMAND ${_RT_PYTHON} ${CMAKE_SOURCE_DIR}/tools/link_smoke_selftest.py
)

# The generated-artifact DRIFT gate (Rule 15 — edit the YAML, never the generated
# files). It existed as `gen_templates.py --check` but was MANUAL ONLY, which left
# a hole underneath runtime_link_smoke above: cq_link_smoke.c is generated, so a
# YAML row added WITHOUT regenerating leaves the smoke table covering 2880 symbols
# while the archive defines 2881. The gate would still pass — every symbol it
# knows about resolves — and the new symbol would be silently unwitnessed,
# reopening exactly the bd 0ktq gap. Running --check in ctest is what makes the
# link gate self-maintaining rather than a snapshot. Read-only w.r.t. the source
# tree: --check regenerates into a tempdir and diffs.
add_test(
    NAME runtime_generated_drift
    COMMAND ${_RT_PYTHON} ${CMAKE_SOURCE_DIR}/tools/gen_templates.py
            --output-dir ${CMAKE_CURRENT_SOURCE_DIR} --check
)

# bd qwcy — the other half of the link witness: the HAND-WRITTEN cqrt_* core ABI.
# runtime_link_smoke above covers only the YAML-GENERATED cq_template_* surface,
# because enumerating a hand-written header from a YAML-driven emitter is the
# wrong layering. This check owns that half and lives outside gen_templates.py
# for exactly that reason.
#
# WHY: cqrt_* is FROZEN-DESIGNATE (the 7.5 cqrt_qram_*, the 7.6 cqrt_tape_*, and
# the kjd1 cqrt_copy_f{16,32,64,80}_controlled families are named freeze gates).
# Every e2e slice already witnesses the COMMON core more strongly than this does
# — the goldens see ARGUMENTS too — so the gap closed here is the RARE half:
# controlled twins and widths no committed slice reaches.
#
# Measured at introduction: 173 declared / 173 defined / 0 mismatch either way.
# Another REGRESSION gate, not a bug-finder.
add_test(
    NAME runtime_core_abi_link
    COMMAND ${_RT_PYTHON} ${CMAKE_SOURCE_DIR}/tools/core_abi_link_check.py
            --archive $<TARGET_FILE:cq_runtime>
)

# Mutant validation. The two-direction (declared vs defined) comparison is
# self-validating BY DESIGN and that is load-bearing: the header is hand-written,
# so an extractor over it is fragile — the first draft used a return-type
# alternation and silently missed `bool cqrt_measure_i1` / `_Float16
# cqrt_measure_f16`, i.e. it UNDER-COVERED while reporting success. M2 pins that
# a defined-but-undeclared symbol is still caught, so an extractor gap can never
# quietly shrink the covered set.
add_test(
    NAME runtime_core_abi_link_selftest
    COMMAND ${_RT_PYTHON} ${CMAKE_SOURCE_DIR}/tools/core_abi_link_check.py
            --archive $<TARGET_FILE:cq_runtime> --selftest
)
